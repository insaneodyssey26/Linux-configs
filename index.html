function nn
    set -l name (test -n "$argv[1]"; and echo "$argv[1]"; or echo "note")
    set -l filename "$name.rnote"

    if test -f ~/Templates/Template.rnote
        cp ~/Templates/Template.rnote "./$filename"
        rnote "./$filename" &
        disown
    else
        echo "Error: Template.rnote not found in ~/Templates/"
    end
end

Setup Script
#!/bin/bash

mkdir -p ~/.config/fish
mkdir -p ~/.config/fastfetch

ln -sf ~/dotfiles/config.fish ~/.config/fish/config.fish
ln -sf ~/dotfiles/fastfetch/config.jsonc ~/.config/fastfetch/config.jsonc

dconf load / < ~/dotfiles/gnome_settings.ini

Installation

$ git clone https://github.com/insaneodyssey26/Linux-configs.git ~/dotfiles
$ cd ~/dotfiles && ./setup/setup.sh

Dependencies

Core: fish, dconf, code-insiders

Fish Tools: eza, bat, fastfetch, zoxide, starship, fzf, rnote

$ sudo pacman -S fish dconf code-insiders eza bat fastfetch zoxide starship fzf rnote
